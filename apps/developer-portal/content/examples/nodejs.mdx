```ts
import { RestClient } from "@thtwaat/rest";

const api = new RestClient({
  apiUrl: process.env.THTWAAT_API_URL!,
  apiKey: process.env.THTWAAT_API_KEY!,
});

const res = await api.agents.chat({ message: "Hello from Node" });
console.log(res);
```
